  }, []);

  const t = translations[lang] || translations['en'];
  const astroData = multiLangAstroContent[formData.reportLang] || multiLangAstroContent['en'];

  const handleInputChange = (e) => {
    setFormData({ ...formData, [e.target.name]: e.target.value });
  };

  const resetFlow = () => {
    setFormData({ name: '', dob: '', tob: '', pob: '', reportLang: lang });
    setPaymentVerified(false); 
    setStep('form');
  };

  const handlePaymentRedirect = (e) => {
    e.preventDefault();
    localStorage.setItem('astro_name', formData.name);
    setStep('processing');
    window.location.href = PAYMENT_LINK;
  };

  return (
    <div className="min-h-screen bg-slate-100 font-sans text-slate-900">
      <header className="bg-white border-b sticky top-0 z-30 print:hidden shadow-sm">
        <div className="max-w-5xl mx-auto px-4 h-16 flex items-center justify-between">
          <div className="flex items-center gap-2">
            <div className="w-10 h-10 bg-indigo-600 rounded-xl flex items-center justify-center text-white">
              <Sparkles size={24} />
            </div>
            <span className="font-bold text-xl tracking-tight">AstroAI Premium</span>
          </div>
          <div className="flex items-center gap-2">
            <Languages size={18} className="text-slate-400" />
            <select 
              value={lang} 
              onChange={(e) => {
                setLang(e.target.value);
                setFormData(prev => ({...prev, reportLang: e.target.value}));
              }}
              className="bg-transparent text-sm font-medium focus:outline-none cursor-pointer"
            >
              <option value="ta">தமிழ்</option>
              <option value="te">తెలుగు</option>
              <option value="kn">ಕನ್ನಡ</option>
              <option value="hi">हिन्दी</option>
              <option value="en">English</option>
            </select>
          </div>
        </div>
      </header>

      <main className="max-w-5xl mx-auto p-4 py-8">
        {step === 'form' && (
          <div className="max-w-4xl mx-auto">
            <div className="text-center mb-10">
              <h1 className="text-3xl md:text-5xl font-extrabold mb-4">{t.title}</h1>
              <p className="text-xl text-slate-600">{t.subtitle}</p>
            </div>
            <div className="bg-white rounded-[2.5rem] shadow-2xl border border-slate-100 overflow-hidden md:flex">
                <div className="md:w-2/5 bg-indigo-600 p-10 text-white relative">
                    <h3 className="text-2xl font-bold mb-6">Lifetime Report</h3>
                    <ul className="space-y-4 text-sm">
                      <li className="flex items-center gap-3"><CheckCircle size={18} /> {t.catHealth} Analysis</li>
                      <li className="flex items-center gap-3"><CheckCircle size={18} /> {t.catWork} Success</li>
                      <li className="flex items-center gap-3"><CheckCircle size={18} /> {t.catWife} Insights</li>
                      <li className="flex items-center gap-3"><CheckCircle size={18} /> {t.catFinance} Growth</li>
                    </ul>
                    <div className="mt-12 bg-white/10 p-4 rounded-2xl">
                      <p className="text-2xl font-black">₹{PACKAGE_PRICE}</p>
                      <p className="text-xs uppercase font-bold">One-Time Access</p>
                    </div>
                </div>
                <div className="md:w-3/5 p-10">
                  <form onSubmit={handlePaymentRedirect} className="space-y-6">
                    <div className="space-y-2">
                      <label className="text-sm font-bold text-slate-700 flex items-center gap-2"><User size={16} /> {t.name}</label>
                      <input required type="text" name="name" value={formData.name} onChange={handleInputChange} className="w-full px-5 py-4 rounded-2xl border border-slate-200 focus:ring-4 focus:ring-indigo-50 outline-none" />
                    </div>
                    <div className="grid grid-cols-2 gap-6">
                      <div className="space-y-2">
                        <label className="text-sm font-bold text-slate-700 flex items-center gap-2"><Calendar size={16} /> {t.dob}</label>
                        <input required type="date" name="dob" value={formData.dob} onChange={handleInputChange} className="w-full px-5 py-4 rounded-2xl border border-slate-200" />
                      </div>
                      <div className="space-y-2">
                        <label className="text-sm font-bold text-slate-700 flex items-center gap-2"><Clock size={16} /> {t.tob}</label>
                        <input required type="time" name="tob" value={formData.tob} onChange={handleInputChange} className="w-full px-5 py-4 rounded-2xl border border-slate-200" />
                      </div>
                    </div>
                    <div className="space-y-2">
                      <label className="text-sm font-bold text-slate-700 flex items-center gap-2"><MapPin size={16} /> {t.pob}</label>
                      <input required type="text" name="pob" value={formData.pob} onChange={handleInputChange} className="w-full px-5 py-4 rounded-2xl border border-slate-200" placeholder="City, State" />
                    </div>
                    <button type="submit" className="w-full bg-indigo-600 hover:bg-indigo-700 text-white font-black text-lg py-5 px-8 rounded-2xl shadow-xl flex items-center justify-center gap-3">
                      <CreditCard size={20} /> {t.btnPay} <ChevronRight />
                    </button>
                  </form>
                </div>
            </div>
          </div>
        )}

        {step === 'processing' && (
          <div className="text-center py-20">
            <div className="w-16 h-16 border-4 border-t-indigo-600 rounded-full animate-spin mx-auto mb-6"></div>
            <h2 className="text-2xl font-bold">{t.processing}</h2>
          </div>
        )}

        {paymentVerified && step === 'result' && (
          <div className="max-w-4xl mx-auto space-y-8">
            <div className="bg-white p-6 rounded-3xl shadow-sm flex justify-between items-center print:hidden">
              <h2 className="text-2xl font-bold text-emerald-600">Verification Success!</h2>
              <div className="flex gap-4">
                <button onClick={resetFlow} className="px-6 py-3 border rounded-xl font-bold flex items-center gap-2">
                  <RefreshCw size={18} /> {t.newReport}
                </button>
                <button onClick={() => window.print()} className="px-6 py-3 bg-indigo-600 text-white rounded-xl font-bold flex items-center gap-2">
                  <Printer size={18} /> {t.print}
                </button>
              </div>
            </div>
            {/* 6-PAGE REPORT RENDERED HERE */}
            <div className="space-y-8 print:space-y-0">
               <div className="report-page bg-white p-12 min-h-[1050px] shadow-xl rounded-3xl">
                  <SectionHeader icon={Sun} title="Personal Profile" pageNumber="1" />
                  <div className="text-center py-20">
                    <h1 className="text-5xl font-black mb-4">{formData.name}</h1>
                    <p className="text-xl text-slate-500 font-mono">{formData.dob} @ {formData.tob}</p>
                    <p className="text-slate-400 mt-2">{formData.pob}</p>
                  </div>
               </div>
               <div className="report-page bg-white p-12 min-h-[1050px] shadow-xl rounded-3xl">
                  <SectionHeader icon={Compass} title="Birth Charts" pageNumber="2" />
                  <div className="grid grid-cols-2 gap-10">
                    <Chart title="Rasi Chart" planets={{ 0: ['ME', 'VE'], 2: ['SU'], 4: ['MA'], 6: ['JU'], 8: ['AS'] }} />
                    <Chart title="Navamsa" planets={{ 1: ['MO'], 3: ['SA'], 5: ['RA'], 7: ['KE'], 0: ['JU'] }} />
                  </div>
               </div>
               <div className="report-page bg-white p-12 min-h-[1050px] shadow-xl rounded-3xl">
                  <SectionHeader icon={BookOpen} title="Life Predictions" pageNumber="3" />
                  <div className="space-y-8">
                    <div><h4 className="font-bold text-indigo-600 uppercase mb-2">{t.catHealth}</h4><p className="text-lg leading-relaxed">{astroData.health}</p></div>
                    <div><h4 className="font-bold text-indigo-600 uppercase mb-2">{t.catWork}</h4><p className="text-lg leading-relaxed">{astroData.work}</p></div>
                  </div>
               </div>
               <div className="report-page bg-white p-12 min-h-[1050px] shadow-xl rounded-3xl">
                  <SectionHeader icon={TrendingUp} title="Finance & Business" pageNumber="4" />
                  <div className="space-y-8">
                    <div><h4 className="font-bold text-indigo-600 uppercase mb-2">{t.catFinance}</h4><p className="text-lg leading-relaxed">{astroData.finance}</p></div>
                    <div><h4 className="font-bold text-indigo-600 uppercase mb-2">{t.catBusiness}</h4><p className="text-lg leading-relaxed">{astroData.business}</p></div>
                  </div>
               </div>
               <div className="report-page bg-white p-12 min-h-[1050px] shadow-xl rounded-3xl">
                  <SectionHeader icon={Heart} title="Family & Partner" pageNumber="5" />
                  <div className="space-y-8">
                    <div><h4 className="font-bold text-rose-600 uppercase mb-2">{t.catWife}</h4><p className="text-lg leading-relaxed">{astroData.wife}</p></div>
                    <div><h4 className="font-bold text-indigo-600 uppercase mb-2">{t.catFamily}</h4><p className="text-lg leading-relaxed">{astroData.family}</p></div>
                  </div>
               </div>
               <div className="report-page bg-white p-12 min-h-[1050px] shadow-xl rounded-3xl">
                  <SectionHeader icon={ShieldAlert} title="Credit & Future" pageNumber="6" />
                  <div className="space-y-8">
                    <div><h4 className="font-bold text-red-600 uppercase mb-2">{t.catCredit}</h4><p className="text-lg leading-relaxed">{astroData.credit}</p></div>
                    <div><h4 className="font-bold text-indigo-600 uppercase mb-2">{t.catFuture}</h4><p className="text-lg leading-relaxed">{astroData.future}</p></div>
                  </div>
                  <div className="mt-auto pt-10 border-t text-center text-slate-400">© 2024 AstroAI Global</div>
               </div>
            </div>
          </div>
        )}

        {(step === 'result' || step === 'generating') && !paymentVerified && (
            <div className="max-w-md mx-auto text-center p-12 bg-white rounded-3xl shadow-xl border border-red-50">
              <Lock size={48} className="mx-auto text-red-600 mb-6" />
              <h2 className="text-2xl font-bold mb-2">Access Locked</h2>
              <p className="text-slate-500 mb-8">Payment of ₹{PACKAGE_PRICE} is required to view this report.</p>
              <button onClick={() => setStep('form')} className="px-8 py-3 bg-indigo-600 text-white rounded-xl font-bold">Try Payment Again</button>
            </div>
        )}
      </main>

      <style>{`
        @media print {
          @page { size: A4; margin: 0; }
          body { background: white !important; }
          .print\\:hidden { display: none !important; }
          .report-page { height: 100vh !important; page-break-after: always !important; box-shadow: none !important; border: none !important; border-radius: 0 !important; }
        }
      `}</style>
    </div>
  );
}
